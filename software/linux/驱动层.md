### 3.交叉编译

~~~C
// 1.链接
https://releases.linaro.org/components/toolchain/binaries/7.2-2017.11/arm-linux-gnueabi/
// 2.将解压的文件复制到/usr/local/arm/中
tar -vxf gcc-linaro-7.2.1-2017.11-x86_64_arm-linux-gnueabi.tar.xz 
// 3.交叉编译器的环境变量
sudo cp  /home/project01/pro01/gcc-linaro-7.2.1-2017.11-x86_64_arm-linux-gnueabi ./
// 4.修改/etc/profile
export PATH=$PATH:/usr/local/arm/gcc-linaro-7.2.1-2017.11-x86_64_arm-linux-gnueabi/bin
// 5.使环境变量生效：
source ~/.bashrc 
// 6.验证是否成功
arm-linux-gnueabi-gcc -v
~~~

~~~c

https://github.com/Lichee-Pi/u-boot

ARCH=arm
CROSS_COMPILE=arm-linux-gnueabi-
// 3.加载配置文件
make licheepi_nano_defconfig
// 4.图形界面配置uboot
make menuconfig
    // 添加
    console=ttyS0,115200 panic=5 rootwait root=/dev/mmcblk0p2 earlyprintk rw
    load mmc 0:1 0x80008000 zImage;load mmc 0:1 0x80c08000 suniv-f1c100s-licheepi-nano.dtb;bootz 0x80008000 - 0x80c08000;
// 5.uboot编译
make -j4
// 6.烧录bin文件
sudo dd if=u-boot-sunxi-with-spl.bin of=/dev/sdb bs=1024 seek=8
~~~
